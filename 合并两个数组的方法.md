# 合并两个数组的小方法

从 C 语言到 C++、Java，对于两个数组的合并，我都学了很多方法，在这里我就简单做个总结，加深印象以便于后面的学习，和应用到解题里去。

## 1. C语言的两个数组合并

### ① 不要求有序（直接上指针，当然硬连也可以）

```c
#include <stdio.h>
#include <stdlib.h>
int main() {
    int n,m;
    scanf("%d %d",&n,&m);
    // 开辟三个空间
    int *arr = (int *) malloc(n * sizeof(int));
    int *brr= (int *) malloc(m * sizeof(int));
    int *crr = (int *) malloc((n+m) * sizeof(int));
    // 记录头指针
    int *a=arr;
    int *b=brr;
    int *c=crr;
    // 初始化
    for (int i = 0; i < n; i++) {
        scanf("%d",a++);
    }
    for (int i = 0; i < m; i++) {
        scanf("%d",b++);
    }
    // 将指针重置，及c的头指针赋值为a，再连b
    a=arr;
    while (a<arr+n) {
        *c++=*a++;
    }
    b=brr;
    while (b<brr+m) {
        *c++=*b++;
    }
    c=crr;
    for (int i = 0; i < n+m; i++) {
        printf("%d ",*c++);
    }
    // 释放空间
    free(arr);
    free(brr);
    free(crr);
    return 0;
}
```

### ② 要求有序（双指针，使用双指针的前提一定要有序）

```c
#include <stdio.h>
#include <stdlib.h>
int compare(const void *a, const void *b) {// 比较函数，用在qsort里的
    return *(int *)a - *(int *)b;
}
int main() {
    int n,m;
    scanf("%d %d",&n,&m);
    // 开辟三个空间
    int *arr = (int *) malloc(n * sizeof(int));
    int *brr= (int *) malloc(m * sizeof(int));
    int *crr = (int *) malloc((n+m) * sizeof(int));
    // 记录头指针
    int *a=arr;
    int *b=brr;
    int *c=crr;
    // 初始化
    for (int i = 0; i < n; i++) {
        scanf("%d",a++);
    }
    for (int i = 0; i < m; i++) {
        scanf("%d",b++);
    }
    // 排序
    qsort(arr,n,sizeof(int),compare);
    qsort(brr,m,sizeof(int),compare);
    a = arr;
    b = brr;
    c = crr;
    // 比大小插入指针
    while (a < arr+n && b < brr+m) {
        if (*a <= *b) {
            *c++ = *a++;
        } else {
            *c++ = *b++;
        }
    }
    // 处理剩余
    while (a < arr+n) *c++ = *a++;
    while (b < brr+m) *c++ = *b++;
    c = crr;
    for (int i = 0; i < n+m; i++) {
        printf("%d ",*c++);
    }
    // 释放空间
    free(arr);
    free(brr);
    free(crr);
    return 0;
}
```

嘻嘻，其实还有一种写法（二分查找优化合并）：

```c
#include <stdio.h>
#include <stdlib.h>
int cmp(const void *a, const void *b) {
    return *(int*)a - *(int*)b;
}
int find(int *arr, int len, int x) {// 查找到第一个 > x 的位置
    int left = 0, right = len;
    while (left < right) {
        int mid = (left + right) / 2;
        if (arr[mid] > x) right = mid;
        else left = mid + 1;
    }
    return left;
}

int main() {
    int n, m;
    scanf("%d %d", &n, &m);
    int *a = (int*)malloc(n * sizeof(int));
    int *b = (int*)malloc(m * sizeof(int));
    int *c = (int*)malloc((n + m) * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }
    for (int i = 0; i < m; i++) {
        scanf("%d", &b[i]);
    }
    qsort(a, n, sizeof(int), cmp);
    qsort(b, m, sizeof(int), cmp);
    int i = 0, j = 0, k = 0;
    while (i < n && j < m) {
        // 在 a 的剩余部分 (a+i) 中查找有多少个元素 <= b[j]
        // find 返回的是相对于 (a+i) 的偏移量
        int pos = find(a + i, n - i, b[j]);
        // 批量复制 a 中小于等于 b[j] 的元素
        for (int p = 0; p < pos; p++) {
            c[k++] = a[i++];
        }
        // 放入当前的 b[j]
        c[k++] = b[j++];
    }
    // 处理剩余
    while (i < n) c[k++] = a[i++];
    while (j < m) c[k++] = b[j++];
    for (int p = 0; p < n + m; p++) {
        printf("%d ", c[p]);
    }
    free(a);
    free(b);
    free(c);
    return 0;
}
```

### ③ 调用 memcpy 方法

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int main() {
    int n,m;
    scanf("%d %d",&n,&m);
    int *arr = (int *) malloc(n * sizeof(int));
    int *brr= (int *) malloc(m * sizeof(int));
    int *crr = (int *) malloc((n+m) * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d",arr+i);
    }
    for (int i = 0; i < m; i++) {
        scanf("%d",brr+i);
    }
    memcpy(crr,arr,n*sizeof(int));
    memcpy(crr+n,brr,m*sizeof(int));
    for (int i = 0; i < n+m; i++) {
        printf("%d ",crr[i]);
    }
    free(arr);
    free(brr);
    free(crr);
    return 0;
}
```

## 2. Java中两个数组的合并

### ① Stream流法

```java
import java.io.*;
import java.util.*;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(System.out);
        
        // 读的时候一定要注意格式！
        String[] s = br.readLine().trim().split(" ");
        int n = Integer.parseInt(s[0]);
        int m = Integer.parseInt(s[1]);
        
        String[] Str1 = br.readLine().trim().split(" ");
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = Integer.parseInt(Str1[i]);
        }
        
        String[] Str2 = br.readLine().trim().split(" ");
        int[] brr = new int[m];
        for (int i = 0; i < m; i++) {
            brr[i] = Integer.parseInt(Str2[i]);
        }
        
        int[] crr = IntStream.concat(Arrays.stream(arr), Arrays.stream(brr)).toArray();
        for (int i = 0; i < crr.length; i++) {
            out.print(crr[i] + " ");
        }
        out.flush();
        br.close();
        out.close();
    }
}
```

### ② System.arraycopy 深拷贝（和 C 里的 memcpy 差不多）

```java
import java.io.*;
import java.util.*;
import static java.lang.System.arraycopy; // 头文件不能少

public class Main {
    public static int[] merge(int[] arr1, int[] arr2) {
        int[] crr = new int[arr1.length + arr2.length];
        arraycopy(arr1, 0, crr, 0, arr1.length);
        arraycopy(arr2, 0, crr, arr1.length, arr2.length);
        return crr;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(System.out);
        
        String[] s = br.readLine().trim().split(" ");
        int n = Integer.parseInt(s[0]);
        int m = Integer.parseInt(s[1]);
        
        String[] Str1 = br.readLine().trim().split(" ");
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = Integer.parseInt(Str1[i]);
        }
        
        String[] Str2 = br.readLine().trim().split(" ");
        int[] brr = new int[m];
        for (int i = 0; i < m; i++) {
            brr[i] = Integer.parseInt(Str2[i]);
        }
        
        int[] crr = merge(arr, brr);
        for (int i = 0; i < crr.length; i++) {
            out.print(crr[i] + " ");
        }
        
        out.flush();
        br.close();
        out.close();
    }
}
```

## 3. C++里的合并方法

### ① 直接使用 Vector 的 insert

```cpp
#include <bits/stdc++.h>
using namespace std;
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n,m;
    cin>>n>>m;
    vector<int> v;
    vector<int> v1;
    for (int i = 0; i < n; i++) {
        int num;
        cin>>num;
        v.push_back(num);
    }
    for (int j=0;j<m;j++) {
        int num1;
        cin>>num1;
        v1.push_back(num1);
    }
    vector<int> merge;
    merge.insert(merge.end(), v.begin(), v.end()); // insert插入合并
    for (auto i:merge) {
        cout<<i<<" ";
    }
    return 0;
}
```

### ② 求并集的 set_union 方法（前提要有序），有去重的作用

```cpp
#include <bits/stdc++.h>
using namespace std;
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n,m;
    cin>>n>>m;
    vector<int> v;
    vector<int> v1;
    for (int i = 0; i < n; i++) {
        int num;
        cin>>num;
        v.push_back(num);
    }
    for (int j=0;j<m;j++) {
        int num1;
        cin>>num1;
        v1.push_back(num1);
    }
    vector<int> merge;
    sort(v.begin(), v.end());
    sort(v1.begin(), v1.end());
    set_union(v.begin(), v.end(), v1.begin(), v1.end(), back_inserter(merge)); // 是STL算法里常用的方法
    for (auto i:merge) {
        cout<<i<<" ";
    }
    return 0;
}
```
